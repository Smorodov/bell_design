double GeometryParams::get_radius_at(double height_pos) const {
    if (!has_profile || profile_radius.empty()) {
        double t = std::max(0.0, std::min(1.0, height_pos));
        return r_min + (r_max - r_min) * t;
    }
    
    double pos = height_pos * (profile_radius.size() - 1);
    size_t idx = static_cast<size_t>(pos);
    double frac = pos - idx;
    
    if (idx >= profile_radius.size() - 1) return profile_radius.back();
    return profile_radius[idx] * (1.0 - frac) + profile_radius[idx + 1] * frac;
}

double GeometryParams::get_thickness_at(double height_pos) const {
    if (!has_profile || profile_thickness.empty()) {
        return thickness;
    }
    
    double pos = height_pos * (profile_thickness.size() - 1);
    size_t idx = static_cast<size_t>(pos);
    double frac = pos - idx;
    
    if (idx >= profile_thickness.size() - 1) return profile_thickness.back();
    return profile_thickness[idx] * (1.0 - frac) + profile_thickness[idx + 1] * frac;
}

bool GeometryParams::is_monotonic() const {
    if (!has_profile || profile_radius.size() < 2) {
        return r_max >= r_min;
    }
    
    for (size_t i = 1; i < profile_radius.size(); ++i) {
        if (profile_radius[i] < profile_radius[i-1]) return false;
    }
    return true;
}

double GeometryParams::get_max_slope() const {
    if (!has_profile || profile_radius.size() < 2) {
        return (r_max - r_min) / height;
    }
    
    double max_slope = 0.0;
    double dh = height / (profile_radius.size() - 1);
    
    for (size_t i = 1; i < profile_radius.size(); ++i) {
        double slope = (profile_radius[i] - profile_radius[i-1]) / dh;
        if (std::abs(slope) > max_slope) max_slope = std::abs(slope);
    }
    return max_slope;
}

double BellSynthesizer::calculate_decay_envelope(double time, const Mode& mode) const {
    double t60 = mode.decay_time;
    if (t60 < 0.001) t60 = 0.001;
    
    double attack_time = mode.attack_time;
    if (attack_time < 0.001) attack_time = 0.001;
    
    double attack = 1.0 - std::exp(-time / attack_time);
    
    double amplitude = mode.amplitude * attack;
    double nonlinear_factor = 1.0 + config_.nonlinear_damping * amplitude * amplitude;
    double effective_t60 = t60 / nonlinear_factor;
    
    double decay = std::exp(-3.0 * time / effective_t60);
    
    double freq_factor = mode.frequency / 1000.0;
    double correction = 1.0 + 0.1 * std::exp(-freq_factor);
    
    double t_rel = time / effective_t60;
    
    if (t_rel < 0.15) {
        double freq_boost = 1.0 + config_.boost_freq_factor * freq_factor;
        double boost_factor = 1.0 + config_.boost_amplitude * freq_boost * (1.0 - t_rel / 0.15);
        decay = decay * boost_factor;
    }
    
    if (t_rel > 0.7) {
        double slowdown = 1.0 + config_.slowdown_factor * ((t_rel - 0.7) / 0.3);
        decay = std::pow(decay, 1.0 / (slowdown * correction));
    }
    
    double envelope = decay * attack;
    
    if (envelope < 1e-12) envelope = 0.0;
    
    return envelope;
}

double BellSynthesizer::get_envelope_value(const EnvelopeData& data, double time) const {
    if (data.envelope.empty()) return 0.0;
    if (time < 0.0) return data.envelope[0];
    if (time >= data.duration) return 0.0;
    
    double idx_double = time / data.step;
    size_t idx = static_cast<size_t>(idx_double);
    double frac = idx_double - idx;
    
    if (idx >= data.envelope.size() - 1) return data.envelope.back();
    
    return data.envelope[idx] * (1.0 - frac) + data.envelope[idx + 1] * frac;
}

std::vector<double> BellSynthesizer::scale_envelope_to_duration(
    const std::vector<double>& original_envelope,
    double original_duration,
    double target_duration,
    const Mode& mode) {
    
    if (original_envelope.empty()) return original_envelope;
    if (target_duration <= 0.0) return original_envelope;
    
    if (std::abs(original_duration - target_duration) < 1e-6) {
        return original_envelope;
    }
    
    // Находим естественную длительность (где огибающая падает ниже порога)
    double peak = mode.amplitude;
    double noise_floor = 1e-8;
    double relative_threshold = config_.envelope_threshold;
    double threshold = std::max(noise_floor, peak * relative_threshold);
    if (threshold < 1e-6) threshold = 1e-6;
    
    bool was_above_threshold = false;
    size_t natural_end = original_envelope.size();
    
    for (size_t i = 0; i < original_envelope.size(); ++i) {
        if (original_envelope[i] > threshold) {
            was_above_threshold = true;
        }
        if (was_above_threshold && original_envelope[i] < threshold) {
            natural_end = i;
            break;
        }
    }
    
    if (natural_end == original_envelope.size()) {
        natural_end = (size_t)(original_envelope.size() * 0.99);
    }
    if (natural_end < original_envelope.size() * 0.1) {
        natural_end = (size_t)(original_envelope.size() * 0.1);
    }
    
    // Масштабируем только активную часть (0 ... natural_end)
    double original_active_duration = (double)natural_end / (original_envelope.size() - 1) * original_duration;
    double scale = target_duration / original_active_duration;
    
    if (original_active_duration < 1e-12) {
        return original_envelope;
    }
    
    std::vector<double> scaled_envelope;
    scaled_envelope.resize(original_envelope.size());
    
    for (size_t i = 0; i < original_envelope.size(); ++i) {
        double t_norm = (double)i / (original_envelope.size() - 1);
        double t_scaled = t_norm / scale;
        
        if (t_scaled >= 1.0) {
            scaled_envelope[i] = 0.0;  // Хвост обнуляем
        } else {
            double pos = t_scaled * (original_envelope.size() - 1);
            size_t idx = (size_t)pos;
            double frac = pos - idx;
            
            if (idx >= original_envelope.size() - 1) {
                scaled_envelope[i] = 0.0;
            } else {
                scaled_envelope[i] = original_envelope[idx] * (1.0 - frac) + 
                                     original_envelope[idx + 1] * frac;
            }
        }
    }
    
    // Энергетическая компенсация НЕ ПРИМЕНЯЕТСЯ
    // Нормализация на выходе выровняет громкость всех файлов
    
    return scaled_envelope;
}

void BellSynthesizer::precompute_envelopes() {
    envelope_cache_.clear();
    
    bool use_natural_duration = (config_.duration < 0.0);
    
    double max_t60 = 0.0;
    for (const auto& mode : modes_) {
        if (mode.decay_time > max_t60) max_t60 = mode.decay_time;
    }
    
    double natural_duration = max_t60 * config_.tail_factor;
    if (natural_duration > 30.0) natural_duration = 30.0;
    if (natural_duration < 1.0) natural_duration = 3.0;
    
    if (use_natural_duration || config_.duration <= 0.0) {
        actual_duration_ = natural_duration;
    } else {
        actual_duration_ = config_.duration;
        if (actual_duration_ < 0.1) actual_duration_ = 0.1;
        if (actual_duration_ > 30.0) actual_duration_ = 30.0;
    }
    
    double table_step = config_.envelope_table_step;
    if (table_step < 0.0001) table_step = 0.0001;
    
    bool need_scaling = !use_natural_duration && 
                        std::abs(natural_duration - actual_duration_) > 1e-6;
    
    fmt::print("   Расчёт огибающих (шаг {:.1f} мс)...\n", table_step * 1000.0);
    if (need_scaling) {
        fmt::print("   Естественная длительность: {:.2f} с\n", natural_duration);
        fmt::print("   Целевая длительность: {:.2f} с (масштабирование)\n", actual_duration_);
    }
    
    for (size_t m = 0; m < modes_.size(); ++m) {
        EnvelopeData data;
        data.step = table_step;
        data.duration = actual_duration_;
        
        if (need_scaling) {
            size_t natural_size = static_cast<size_t>(natural_duration / table_step) + 1;
            if (natural_size < 10) natural_size = 10;
            
            std::vector<double> natural_envelope(natural_size);
            for (size_t i = 0; i < natural_size; ++i) {
                double t = i * table_step;
                natural_envelope[i] = calculate_decay_envelope(t, modes_[m]);
            }
            
            data.envelope = scale_envelope_to_duration(
                natural_envelope, natural_duration, actual_duration_, modes_[m]);
        } else {
            size_t table_size = static_cast<size_t>(actual_duration_ / table_step) + 1;
            if (table_size < 10) table_size = 10;
            data.envelope.resize(table_size);
            for (size_t i = 0; i < table_size; ++i) {
                double t = i * table_step;
                data.envelope[i] = calculate_decay_envelope(t, modes_[m]);
            }
        }
        
        envelope_cache_[m] = std::move(data);
    }
    
    fmt::print("   Огибающие рассчитаны: {} мод\n", modes_.size());
    fmt::print("   Длительность: {:.2f} с", actual_duration_);
    if (use_natural_duration) {
        fmt::print(" (естественная)");
    } else if (need_scaling) {
        fmt::print(" (масштабирована из {:.2f} с)", natural_duration);
    } else {
        fmt::print(" (фиксированная)");
    }
    fmt::print("\n");
}

void BellSynthesizer::add_mode_to_audio(const Mode& mode, 
                                        const EnvelopeData& envelope_data,
                                        std::vector<double>& buffer) {
    if (envelope_data.envelope.empty()) return;
    
    double dt = 1.0 / (config_.sample_rate * config_.oversample_factor);
    size_t max_samples = std::min(buffer.size(), 
                                  static_cast<size_t>(envelope_data.duration / dt));
    double amp = mode.amplitude;
    double freq = mode.frequency;
    
    double nyquist = 0.5 / dt;
    if (freq >= nyquist) {
        fmt::print(stderr, "Warning: частота {:.1f} Гц превышает Найквиста ({:.1f} Гц), пропускаем моду\n", 
                   freq, nyquist);
        return;
    }
    
    for (size_t i = 0; i < max_samples; ++i) {
        double t = i * dt;
        double envelope = get_envelope_value(envelope_data, t);
        double oscillation = std::sin(2.0 * M_PI * mode.frequency * t + mode.phase);
        buffer[i] += amp * envelope * oscillation;
    }
}

void BellSynthesizer::synthesize() {
    if (modes_.empty()) {
        fmt::print(stderr, "Ошибка: нет мод для синтеза\n");
        return;
    }
    
    precompute_envelopes();
    
    size_t total_samples_os = static_cast<size_t>(actual_duration_ * 
                                                   config_.sample_rate * 
                                                   config_.oversample_factor);
    if (total_samples_os < 1) total_samples_os = 1;
    
    audio_oversampled_.assign(total_samples_os, 0.0);
    
    fmt::print("   Синтез звука ({:.2f} с, {} Гц, x{} oversample)...\n", 
               actual_duration_, config_.sample_rate, config_.oversample_factor);
    
    for (size_t m = 0; m < modes_.size(); ++m) {
        auto it = envelope_cache_.find(m);
        if (it != envelope_cache_.end()) {
            add_mode_to_audio(modes_[m], it->second, audio_oversampled_);
        }
    }
    
    size_t total_samples = static_cast<size_t>(actual_duration_ * config_.sample_rate);
    if (total_samples < 1) total_samples = 1;
    
    audio_.assign(total_samples, 0.0);
    
    for (size_t i = 0; i < total_samples; ++i) {
        double sum = 0.0;
        size_t start = i * config_.oversample_factor;
        size_t end = std::min(start + config_.oversample_factor, total_samples_os);
        
        for (size_t j = start; j < end; ++j) {
            sum += audio_oversampled_[j];
        }
        audio_[i] = sum / (end - start);
    }
    
    if (config_.use_acoustic_model) {
        apply_acoustic_coupling();
        if (config_.use_helmholtz_resonance) {
            apply_helmholtz_resonance();
        }
        apply_radiation_filter();
    }
    
    if (config_.use_mount_model) {
        apply_mount_filter();
    }
    
    fmt::print("   Синтез завершён\n");
}

void BellSynthesizer::apply_acoustic_coupling() {
    if (!config_.use_acoustic_coupling) return;
    
    size_t total_samples = audio_.size();
    double dt = 1.0 / config_.sample_rate;
    
    for (const auto& mode : modes_) {
        if (mode.frequency > config_.max_frequency) continue;
        
        double amp = mode.amplitude * 0.05;
        double freq = mode.frequency;
        double damping = 0.01;
        
        for (size_t i = 0; i < total_samples; ++i) {
            double t = i * dt;
            double envelope = std::exp(-damping * t);
            double osc = std::sin(2.0 * M_PI * freq * t + mode.phase);
            audio_[i] += amp * envelope * osc;
        }
    }
}

void BellSynthesizer::apply_helmholtz_resonance() {
    if (!config_.use_helmholtz_resonance) return;
    
    size_t total_samples = audio_.size();
    double dt = 1.0 / config_.sample_rate;
    
    double V = M_PI * geometry_.r_max * geometry_.r_max * geometry_.height;
    double A = M_PI * geometry_.r_max * geometry_.r_max;
    double L = geometry_.height * 0.1;
    double c = 343.0;
    
    double f_h = (c / (2.0 * M_PI)) * std::sqrt(A / (V * L));
    f_h = std::max(f_h, 50.0);
    f_h = std::min(f_h, 500.0);
    
    double amp = config_.helmholtz_amplitude;
    double damping = config_.helmholtz_damping;
    
    for (size_t i = 0; i < total_samples; ++i) {
        double t = i * dt;
        double envelope = std::exp(-damping * t);
        double osc = std::sin(2.0 * M_PI * f_h * t);
        audio_[i] += amp * envelope * osc;
    }
}

void BellSynthesizer::apply_radiation_filter() {
    if (!config_.use_acoustic_model) return;
    
    double efficiency = config_.radiation_efficiency;
    double damping = config_.radiation_damping_factor;
    
    size_t total_samples = audio_.size();
    double dt = 1.0 / config_.sample_rate;
    
    for (size_t i = 0; i < total_samples; ++i) {
        double t = i * dt;
        double filter = 1.0 + efficiency * std::exp(-damping * t);
        audio_[i] *= filter;
    }
}

void BellSynthesizer::apply_mount_filter() {
    if (!config_.use_mount_model) return;
    
    size_t total_samples = audio_.size();
    double dt = 1.0 / config_.sample_rate;
    double damping = config_.mount_damping;
    
    double fc = config_.target_fundamental * 0.5;
    double alpha = 1.0 / (1.0 + 2.0 * M_PI * fc * dt);
    
    double prev = 0.0;
    for (size_t i = 0; i < total_samples; ++i) {
        double sample = audio_[i];
        double filtered = alpha * sample + (1.0 - alpha) * prev;
        audio_[i] = filtered * (1.0 - damping * (1.0 - filtered / (sample + 1e-10)));
        prev = filtered;
    }
}

void BellSynthesizer::add_reverb() {
    if (config_.reverb_mix < 0.001) return;
    
    size_t total_samples = audio_.size();
    
    int delay_samples = static_cast<int>(config_.reverb_delay * config_.sample_rate);
    double feedback = config_.reverb_feedback;
    double mix = config_.reverb_mix;
    
    if (delay_samples >= static_cast<int>(total_samples)) {
        delay_samples = static_cast<int>(total_samples * 0.05);
    }
    
    std::vector<double> reverb(total_samples, 0.0);
    
    for (size_t i = delay_samples; i < total_samples; ++i) {
        reverb[i] = audio_[i - delay_samples] * 0.7 + 
                    reverb[i - delay_samples] * feedback * 0.3;
    }
    
    for (size_t i = 0; i < total_samples; ++i) {
        audio_[i] = audio_[i] * (1.0 - mix) + reverb[i] * mix;
    }
}

void BellSynthesizer::apply_fade(double fade_in_time, double fade_out_time) {
    size_t total_samples = audio_.size();
    
    size_t fade_in_samples = static_cast<size_t>(fade_in_time * config_.sample_rate);
    size_t fade_out_samples = static_cast<size_t>(fade_out_time * config_.sample_rate);
    
    for (size_t i = 0; i < fade_in_samples && i < total_samples; ++i) {
        double t = static_cast<double>(i) / fade_in_samples;
        double fade = t * t * (3.0 - 2.0 * t);
        audio_[i] *= fade;
    }
    
    for (size_t i = 0; i < fade_out_samples && i < total_samples; ++i) {
        size_t idx = total_samples - 1 - i;
        double t = static_cast<double>(i) / fade_out_samples;
        double fade = 1.0 - t * t * (3.0 - 2.0 * t);
        audio_[idx] *= fade;
    }
}

void BellSynthesizer::normalize(double peak_level) {
    if (audio_.empty()) return;
    
    peak_amplitude_ = 0.0;
    double rms_sum = 0.0;
    
    for (double sample : audio_) {
        double abs_val = std::abs(sample);
        if (abs_val > peak_amplitude_) peak_amplitude_ = abs_val;
        rms_sum += sample * sample;
    }
    
    rms_amplitude_ = std::sqrt(rms_sum / audio_.size());
    
    if (peak_amplitude_ < 1e-10) return;
    
    double gain = peak_level / peak_amplitude_;
    
    if (config_.use_rms_normalization && rms_amplitude_ > 1e-10) {
        double rms_gain = config_.target_rms / rms_amplitude_;
        gain = std::min(gain, rms_gain * 1.2);
    }
    
    for (double& sample : audio_) {
        sample *= gain;
    }
}

bool BellSynthesizer::save_wav(const std::string& filename) {
    if (audio_.empty()) {
        fmt::print(stderr, "Ошибка: нет аудио для сохранения\n");
        return false;
    }
    
    std::ofstream file(filename, std::ios::binary);
    if (!file) {
        fmt::print(stderr, "Ошибка: не удалось создать файл {}\n", filename);
        return false;
    }
    
    std::vector<int16_t> pcm_audio(audio_.size());
    for (size_t i = 0; i < audio_.size(); ++i) {
        double sample = std::max(-1.0, std::min(1.0, audio_[i]));
        pcm_audio[i] = static_cast<int16_t>(sample * 32767.0);
    }
    
    int data_size = static_cast<int>(pcm_audio.size() * sizeof(int16_t));
    
    write_wav_header(file, data_size);
    
    file.write(reinterpret_cast<const char*>(pcm_audio.data()), data_size);
    
    if (!file.good()) {
        fmt::print(stderr, "Ошибка: запись в файл не удалась\n");
        file.close();
        return false;
    }
    
    file.close();
    
    fmt::print("   Сохранён: {}\n", filename);
    fmt::print("   Размер: {} KB\n", data_size / 1024);
    
    return true;
}

void BellSynthesizer::write_wav_header(std::ofstream& file, int data_size) const {
    int sample_rate = config_.sample_rate;
    int channels = 1;
    int bits_per_sample = 16;
    
    int byte_rate = sample_rate * channels * bits_per_sample / 8;
    int block_align = channels * bits_per_sample / 8;
    int subchunk2_size = data_size;
    int chunk_size = 36 + subchunk2_size;
    
    file.write("RIFF", 4);
    file.write(reinterpret_cast<const char*>(&chunk_size), 4);
    file.write("WAVE", 4);
    file.write("fmt ", 4);
    
    int subchunk1_size = 16;
    short audio_format = 1;
    short num_channels = static_cast<short>(channels);
    int sample_rate_int = sample_rate;
    short bits_per_sample_short = static_cast<short>(bits_per_sample);
    
    file.write(reinterpret_cast<const char*>(&subchunk1_size), 4);
    file.write(reinterpret_cast<const char*>(&audio_format), 2);
    file.write(reinterpret_cast<const char*>(&num_channels), 2);
    file.write(reinterpret_cast<const char*>(&sample_rate_int), 4);
    file.write(reinterpret_cast<const char*>(&byte_rate), 4);
    file.write(reinterpret_cast<const char*>(&block_align), 2);
    file.write(reinterpret_cast<const char*>(&bits_per_sample_short), 2);
    
    file.write("data", 4);
    file.write(reinterpret_cast<const char*>(&subchunk2_size), 4);
}

void BellSynthesizer::analyze_spectrum(const std::vector<double>& frequencies) {
    if (audio_.empty()) return;
    
    size_t n = audio_.size();
    double dt = 1.0 / config_.sample_rate;
    
    fmt::print("\n🎵 СПЕКТР СИНТЕЗИРОВАННОГО ЗВУКА:\n");
    fmt::print("   {:-<60}\n", "");
    fmt::print("   Основные частоты (Гц):\n");
    fmt::print("   {:-<60}\n", "");
    
    double time_norm = 1.0 / n;
    
    for (double freq : frequencies) {
        if (freq > config_.max_frequency) continue;
        
        double real = 0.0, imag = 0.0;
        for (size_t i = 0; i < n; ++i) {
            double t = i * dt;
            double phase = 2.0 * M_PI * freq * t;
            real += audio_[i] * std::cos(phase);
            imag += audio_[i] * std::sin(phase);
        }
        double amp = std::sqrt(real * real + imag * imag) * time_norm;
        
        if (amp > 0.0001) {
            fmt::print("   {:.1f} Гц : {:.3f} мВ\n", freq, amp * 1000.0);
        }
    }
    
    fmt::print("   {:-<60}\n", "");
}

void BellSynthesizer::print_spectrum(const std::vector<Mode>& modes) const {
    if (modes.empty()) return;
    
    fmt::print("\n🎵 СПЕКТР МОД:\n");
    fmt::print("   {:-<60}\n", "");
    fmt::print("   № | Частота(Гц) | T60(с) | Атака(с) | Амплитуда\n");
    fmt::print("   {:-<60}\n", "");
    
    for (size_t i = 0; i < modes.size() && i < 25; ++i) {
        const auto& m = modes[i];
        fmt::print("   {:2d} | {:10.1f} | {:6.1f} | {:7.3f} | {:8.6f}\n",
                   i+1, m.frequency, m.decay_time, m.attack_time, m.amplitude);
    }
    fmt::print("   {:-<60}\n", "");
}

void BellSynthesizer::analyze_fft(int fft_size) {
    if (audio_.empty()) {
        fmt::print(stderr, "Ошибка: нет аудио для FFT анализа\n");
        return;
    }

#ifndef USE_FFTW
    fmt::print(stderr, "Ошибка: FFTW не доступен. Пересоберите проект с USE_FFTW\n");
    return;
#endif

    size_t max_fft_size = 65536;
    size_t target = (fft_size <= 0) ? std::min(audio_.size(), max_fft_size) 
                                    : std::min(static_cast<size_t>(fft_size), 
                                               std::min(audio_.size(), max_fft_size));
    
    size_t n = 1;
    while (n < target) n <<= 1;

    if (n < 2) {
        fmt::print(stderr, "Ошибка: размер FFT слишком мал (n < 2)\n");
        return;
    }

    fmt::print("   Выполняется FFT (размер: {}, разрешение: {:.1f} Гц)...\n", 
               n, (config_.sample_rate / n));

#ifdef USE_FFTW
    fftw_complex* in = (fftw_complex*)fftw_malloc(sizeof(fftw_complex) * n);
    fftw_complex* out = (fftw_complex*)fftw_malloc(sizeof(fftw_complex) * n);
    
    if (!in || !out) {
        fmt::print(stderr, "Ошибка: не удалось выделить память для FFT\n");
        if (in) fftw_free(in);
        if (out) fftw_free(out);
        return;
    }

    for (size_t i = 0; i < n; ++i) {
        double window = 0.5 * (1.0 - std::cos(2.0 * M_PI * i / (n - 1)));
        if (i < target) {
            in[i][0] = audio_[i] * window;
            in[i][1] = 0.0;
        } else {
            in[i][0] = 0.0;
            in[i][1] = 0.0;
        }
    }

    fftw_plan plan = fftw_plan_dft_1d((int)n, in, out, FFTW_FORWARD, FFTW_ESTIMATE);
    fftw_execute(plan);

    double dt = 1.0 / config_.sample_rate;
    fft_frequencies_.clear();
    fft_magnitudes_.clear();
    
    double window_compensation = 2.0;
    
    for (size_t i = 0; i < n / 2; ++i) {
        double freq = i / (n * dt);
        double mag = std::sqrt(out[i][0] * out[i][0] + out[i][1] * out[i][1]) / n * window_compensation;
        fft_frequencies_.push_back(freq);
        fft_magnitudes_.push_back(mag);
    }

    fftw_destroy_plan(plan);
    fftw_free(in);
    fftw_free(out);
#endif
}

void BellSynthesizer::print_fft_peaks(const std::vector<double>& target_frequencies) {
    if (fft_frequencies_.empty()) {
        fmt::print(stderr, "Сначала выполните analyze_fft()\n");
        return;
    }

    fmt::print("\n📊 СПЕКТР (ТОЛЬКО ЗАДАННЫЕ ЧАСТОТЫ):\n");
    fmt::print("   {:-<60}\n", "");
    fmt::print("   Частота(Гц) | Амплитуда\n");
    fmt::print("   {:-<60}\n", "");

    double freq_resolution = (fft_frequencies_.size() > 1) ? 
                             (fft_frequencies_[1] - fft_frequencies_[0]) : 0.5;
    double tolerance = std::max(0.5, freq_resolution * 3.0);
    
    for (double target_freq : target_frequencies) {
        if (target_freq < 1.0) continue;
        
        double best_freq = 0.0;
        double best_mag = 0.0;
        
        for (size_t i = 0; i < fft_frequencies_.size(); ++i) {
            double diff = std::abs(fft_frequencies_[i] - target_freq);
            if (diff < tolerance && fft_magnitudes_[i] > best_mag) {
                best_freq = fft_frequencies_[i];
                best_mag = fft_magnitudes_[i];
            }
        }
        
        if (best_mag > 0.0) {
            fmt::print("   {:.2f} Гц -> {:.6f} (найдено {:.2f} Гц)\n", 
                       target_freq, best_mag, best_freq);
        } else {
            fmt::print("   {:.2f} Гц -> НЕ НАЙДЕНА\n", target_freq);
        }
    }
    
    fmt::print("   {:-<60}\n", "");
}

void BellSynthesizer::save_spectrum_csv(const std::string& filename, 
                                        const std::vector<double>& target_frequencies) {
    if (fft_frequencies_.empty()) {
        fmt::print(stderr, "Сначала выполните analyze_fft()\n");
        return;
    }

    std::ofstream file(filename);
    if (!file.is_open()) {
        fmt::print(stderr, "Ошибка: не удалось создать {}\n", filename);
        return;
    }

    file << "TargetFrequency(Hz),FoundFrequency(Hz),Magnitude\n";
    
    double freq_resolution = (fft_frequencies_.size() > 1) ? 
                             (fft_frequencies_[1] - fft_frequencies_[0]) : 0.5;
    double tolerance = std::max(0.5, freq_resolution * 3.0);
    int found_count = 0;
    
    for (double target_freq : target_frequencies) {
        if (target_freq < 1.0) continue;
        
        double best_freq = 0.0;
        double best_mag = 0.0;
        
        for (size_t i = 0; i < fft_frequencies_.size(); ++i) {
            double diff = std::abs(fft_frequencies_[i] - target_freq);
            if (diff < tolerance && fft_magnitudes_[i] > best_mag) {
                best_freq = fft_frequencies_[i];
                best_mag = fft_magnitudes_[i];
            }
        }
        
        if (best_mag > 0.0) {
            file << std::fixed << std::setprecision(2) 
                 << target_freq << ","
                 << std::setprecision(2) << best_freq << ","
                 << std::scientific << std::setprecision(6) << best_mag << "\n";
            found_count++;
        } else {
            file << std::fixed << std::setprecision(2) 
                 << target_freq << ",NOT_FOUND,0.0\n";
        }
    }
    
    file.close();
    
    fmt::print("   Спектр сохранён: {} ({} из {} частот найдено)\n", 
               filename, found_count, target_frequencies.size());
}

void BellSynthesizer::save_all_envelopes_csv(const std::string& filename) {
    if (envelope_cache_.empty()) {
        fmt::print(stderr, "Ошибка: огибающие ещё не рассчитаны\n");
        return;
    }

    std::ofstream file(filename);
    if (!file.is_open()) {
        fmt::print(stderr, "Ошибка: не удалось создать {}\n", filename);
        return;
    }

    double dt = 1.0 / (config_.sample_rate * config_.oversample_factor);
    size_t total_samples = static_cast<size_t>(actual_duration_ * 
                                                config_.sample_rate * 
                                                config_.oversample_factor);
    double step = 0.01;
    int step_samples = static_cast<int>(step / dt);
    if (step_samples < 1) step_samples = 1;

    file << "Time(ms)";
    for (size_t i = 0; i < modes_.size(); ++i) {
        file << ",Mode" << (i+1) << "_" << std::fixed << std::setprecision(1) << modes_[i].frequency << "Hz";
    }
    file << "\n";

    for (int i = 0; i < (int)total_samples; i += step_samples) {
        double t = i * dt;
        file << std::fixed << std::setprecision(3) << (t * 1000.0);
        
        for (size_t m = 0; m < modes_.size(); ++m) {
            auto it = envelope_cache_.find(m);
            if (it != envelope_cache_.end()) {
                double envelope = get_envelope_value(it->second, t);
                file << "," << std::setprecision(6) << envelope;
            } else {
                file << ",0.0";
            }
        }
        file << "\n";
    }
    
    file.close();
    fmt::print("   Все огибающие сохранены: {}\n", filename);
}

// ============================================================
// ВСПОМОГАТЕЛЬНАЯ СТРУКТУРА ДЛЯ ИНТЕГРАТОРА
// ============================================================
struct IntegrandParams {
    const BellGenerator* generator;
};

static double integrand_wrapper_parametric(double t, void* params) {
    IntegrandParams* p = static_cast<IntegrandParams*>(params);
    return p->generator->integrandParametric(t);
}

// ============================================================
// КОНСТРУКТОР
// ============================================================
BellGenerator::BellGenerator(const BellMaterial& material, const BellMeshConfig& config)
    : material_(material)
    , config_(config)
    , spline_(nullptr)
    , acc_(nullptr)
    , integ_w_(nullptr)
{
    acc_ = gsl_interp_accel_alloc();
    if (!acc_) {
        logError("Failed to allocate GSL accelerator");
    }
    
    integ_w_ = gsl_integration_workspace_alloc(10000);
    if (!integ_w_) {
        logError("Failed to allocate integration workspace");
    }
}

// ============================================================
// ДЕСТРУКТОР
// ============================================================
BellGenerator::~BellGenerator() {
    if (spline_) {
        gsl_spline_free(spline_);
        spline_ = nullptr;
    }
    if (acc_) {
        gsl_interp_accel_free(acc_);
        acc_ = nullptr;
    }
    if (integ_w_) {
        gsl_integration_workspace_free(integ_w_);
        integ_w_ = nullptr;
    }
}

// ============================================================
// ПОСТРОЕНИЕ СПЛАЙНА
// ============================================================
bool BellGenerator::buildSpline() {
    if (spline_) {
        gsl_spline_free(spline_);
        spline_ = nullptr;
    }
    
    size_t n = geometry_.r_knots.size();
    if (n < 3) {
        logError("At least 3 knots required");
        return false;
    }
    
    std::vector<double> z_knots(n);
    std::vector<double> r_knots(n);
    
    for (size_t i = 0; i < n; ++i) {
        z_knots[i] = geometry_.getKnotZ(i);
        r_knots[i] = geometry_.r_knots[i] * geometry_.R_mouth;
    }
    
    if (std::abs(z_knots.front()) > 1e-6 || std::abs(r_knots.front()) > 1e-6) {
        logError("First knot must be (0, 0)");
        return false;
    }
    if (std::abs(z_knots.back() - geometry_.H) > 1e-6 || 
        std::abs(r_knots.back() - geometry_.R_mouth) > 1e-6) {
        logError("Last knot must be (H, R_mouth)");
        return false;
    }
    
    for (size_t i = 1; i < n; ++i) {
        if (z_knots[i] <= z_knots[i-1]) {
            logError("z_knots must be strictly increasing");
            return false;
        }
    }
    
    for (size_t i = 1; i < n; ++i) {
        if (r_knots[i] <= r_knots[i-1]) {
            logError("r_knots must be strictly increasing");
            return false;
        }
    }
    
    spline_ = gsl_spline_alloc(gsl_interp_cspline, n);
    if (!spline_) {
        logError("Failed to allocate GSL spline");
        return false;
    }
    
    int status = gsl_spline_init(spline_, z_knots.data(), r_knots.data(), n);
    if (status != GSL_SUCCESS) {
        logError("Failed to initialize GSL spline");
        gsl_spline_free(spline_);
        spline_ = nullptr;
        return false;
    }
    
    return true;
}

// ============================================================
// ИНТЕГРАНТ ДЛЯ ДЛИНЫ ДУГИ
// ============================================================
double BellGenerator::integrandParametric(double t) const {
    if (!spline_) return 0.0;
    double z = t * geometry_.H;
    double dr_dz = derivative(z);
    double dz_dt = geometry_.H;
    double dr_dt = dr_dz * geometry_.H;
    return std::sqrt(dz_dt * dz_dt + dr_dt * dr_dt);
}

// ============================================================
// ВЫЧИСЛЕНИЕ ДЛИНЫ ДУГИ
// ============================================================
double BellGenerator::computeArcLengthToT(double t_target) const {
    if (!integ_w_) {
        logError("Integration workspace not initialized");
        return 0.0;
    }
    
    if (t_target <= 0.0) return 0.0;
    
    double t_clamped = std::max(0.0, std::min(t_target, 1.0));
    if (t_clamped < 1e-12) return 0.0;
    
    IntegrandParams params;
    params.generator = this;
    
    gsl_function F;
    F.function = integrand_wrapper_parametric;
    F.params = &params;
    
    double result = 0.0;
    double error = 0.0;
    
    gsl_set_error_handler_off();
    
    int status = gsl_integration_qags(
        &F, 
        0.0, 
        t_clamped, 
        0.0,
        1e-6,
        10000,
        integ_w_,
        &result,
        &error
    );
    
    gsl_set_error_handler(NULL);
    
    if (status != GSL_SUCCESS) {
        logError("Integration failed");
        return 0.0;
    }
    
    if (result < 0.0) result = 0.0;
    return result;
}

double BellGenerator::computeArcLength() {
    if (!spline_) {
        logError("Spline not initialized");
        return 0.0;
    }
    
    if (geometry_.H <= 0.0) {
        logError("H must be > 0");
        return 0.0;
    }
    
    double total = computeArcLengthToT(1.0);
    if (total < 1e-6) {
        logError("Arc length calculation failed - got zero");
        return 0.0;
    }
    
    mesh_data_.total_arc_length = total;
    return total;
}

// ============================================================
// ПОИСК Z ПО ДЛИНЕ ДУГИ
// ============================================================
double BellGenerator::getZfromS(double s) const {
    if (s <= 0.0) return 0.0;
    if (s >= mesh_data_.total_arc_length) return geometry_.H;
    
    double t_low = 0.0;
    double t_high = 1.0;
    
    for (int i = 0; i < 100; ++i) {
        double t_mid = (t_low + t_high) / 2.0;
        double s_mid = computeArcLengthToT(t_mid);
        
        if (s_mid < s) {
            t_low = t_mid;
        } else {
            t_high = t_mid;
        }
        
        if (t_high - t_low < 1e-10) break;
    }
    
    double t = (t_low + t_high) / 2.0;
    return t * geometry_.H;
}

// ============================================================
// РАДИУС, ПРОИЗВОДНАЯ, ТОЛЩИНА
// ============================================================
double BellGenerator::radius(double z) const {
    if (!spline_) return 0.0;
    if (z <= 0.0) return 0.0;
    if (z >= geometry_.H) return geometry_.R_mouth;
    return gsl_spline_eval(spline_, z, acc_);
}

double BellGenerator::derivative(double z) const {
    if (!spline_) return 0.0;
    if (z <= 0.0) return 0.0;
    if (z >= geometry_.H) {
        return gsl_spline_eval_deriv(spline_, geometry_.H, acc_);
    }
    return gsl_spline_eval_deriv(spline_, z, acc_);
}

double BellGenerator::getThicknessAt(double z) const {
    if (z <= 0.0) return geometry_.t_top;
    if (z >= geometry_.H) return geometry_.t_bottom;
    double t = z / geometry_.H;
    return geometry_.t_top + (geometry_.t_bottom - geometry_.t_top) * t;
}

// ============================================================
// ВНЕШНИЙ И ВНУТРЕННИЙ РАДИУСЫ
// ============================================================
double BellGenerator::getOuterRadius(double z) const {
    if (z < 0.0 || z > geometry_.H) return 0.0;
    return radius(z);
}

double BellGenerator::getInnerRadius(double z) const {
    if (z < 0.0 || z > geometry_.H) return 0.0;
    if (z <= 0.0) return 0.0;
    double r = radius(z);
    double dr = derivative(z);
    double len = std::sqrt(1.0 + dr * dr);
    double t = getThicknessAt(z);
    double r_inner = r - t * dr / len;
    return std::max(0.0, r_inner);
}

// ============================================================
// ПОСТРОЕНИЕ ВНЕШНЕГО ПРОФИЛЯ
// ============================================================
void BellGenerator::computeOuterProfile() {
    int n = config_.nNodes;
    mesh_data_.outer_profile.clear();
    mesh_data_.outer_profile.reserve(n);
    
    for (int i = 0; i < n; ++i) {
        double s_target = i * mesh_data_.total_arc_length / (n - 1);
        double z = getZfromS(s_target);
        double r = radius(z);
        mesh_data_.outer_profile.push_back({r, z});
    }
}

// ============================================================
// ВЫЧИСЛЕНИЕ НОРМАЛЕЙ (ГАРАНТИРОВАННО ВНУТРЬ И ВНИЗ)
// ============================================================
void BellGenerator::computeNormals(std::vector<double>& n_r, std::vector<double>& n_z) const {
    int n = config_.nNodes;
    n_r.resize(n);
    n_z.resize(n);
    
    for (int i = 0; i < n; ++i) {
        double z = mesh_data_.outer_profile[i].z;
        double r = mesh_data_.outer_profile[i].r;
        
        if (i == 0 || (r < 1e-12 && z < 1e-12)) {
            n_r[i] = 0.0;
            n_z[i] = 1.0;
            continue;
        }
        
        if (r < 1e-12) {
            n_r[i] = 0.0;
            n_z[i] = 1.0;
            continue;
        }
        
        double dr = derivative(z);
        double len = std::sqrt(1.0 + dr * dr);
        
        n_r[i] = -std::abs(dr) / len;
        n_z[i] = 1.0 / len;
        
        double final_len = std::sqrt(n_r[i] * n_r[i] + n_z[i] * n_z[i]);
        if (final_len > 1e-12) {
            n_r[i] /= final_len;
            n_z[i] /= final_len;
        }
    }
}

// ============================================================
// ПОСТРОЕНИЕ ВНУТРЕННЕГО ПРОФИЛЯ
// ============================================================
void BellGenerator::computeInnerProfile(const std::vector<double>& n_r, const std::vector<double>& n_z) {
    int n = config_.nNodes;
    mesh_data_.inner_profile.clear();
    mesh_data_.inner_profile.reserve(n);
    
    for (int i = 0; i < n; ++i) {
        double z_outer = mesh_data_.outer_profile[i].z;
        double r_outer = mesh_data_.outer_profile[i].r;
        double t = getThicknessAt(z_outer);
        
        double r_inner, z_inner;
        if (i == 0) {
            r_inner = 0.0;
            z_inner = geometry_.t_top;
        } else {
            r_inner = r_outer + t * n_r[i];
            z_inner = z_outer + t * n_z[i];
            if (r_inner < 0.0) r_inner = 0.0;
        }
        
        mesh_data_.inner_profile.push_back({r_inner, z_inner});
    }
}

// ============================================================
// СБОРКА СЕТКИ
// ============================================================
void BellGenerator::assembleMesh() {
    int n = config_.nNodes;
    mesh_data_.nodes.clear();
    mesh_data_.nodes.reserve(2 * n);
    mesh_data_.elements.clear();
    mesh_data_.elements.reserve(n - 1);
    
    for (int i = 0; i < n; ++i) {
        Node node;
        node.id = i + 1;
        node.r = mesh_data_.outer_profile[i].r;
        node.z = mesh_data_.outer_profile[i].z;
        mesh_data_.nodes.push_back(node);
    }
    
    for (int i = 0; i < n; ++i) {
        Node node;
        node.id = i + 1 + n;
        node.r = mesh_data_.inner_profile[i].r;
        node.z = mesh_data_.inner_profile[i].z;
        mesh_data_.nodes.push_back(node);
    }
    
    for (int i = 0; i < n - 1; ++i) {
        Element elem;
        elem.n1 = i + 1 + n;
        elem.n2 = i + 1;
        elem.n3 = i + 2;
        elem.n4 = i + 2 + n;
        mesh_data_.elements.push_back(elem);
    }
}

// ============================================================
// ВЫЧИСЛЕНИЕ МЕТРИК
// ============================================================
void BellGenerator::computeMetrics() {
    double min_t = std::numeric_limits<double>::max();
    double max_t = 0.0;
    
    for (const auto& p : mesh_data_.outer_profile) {
        double t = getThicknessAt(p.z);
        if (t < min_t) min_t = t;
        if (t > max_t) max_t = t;
    }
    
    mesh_data_.min_thickness = min_t;
    mesh_data_.max_thickness = max_t;
}

// ============================================================
// ГЕОМЕТРИЧЕСКИЕ ПРОВЕРКИ
// ============================================================
bool BellGenerator::checkMonotonicity() const {
    const auto& r = geometry_.r_knots;
    for (size_t i = 1; i < r.size(); ++i) {
        if (r[i] <= r[i-1]) return false;
    }
    return true;
}

bool BellGenerator::checkSelfIntersection() const {
    const auto& inner = mesh_data_.inner_profile;
    if (inner.size() < 2) return true;
    
    for (size_t i = 1; i < inner.size(); ++i) {
        double dx = inner[i].r - inner[i-1].r;
        double dz = inner[i].z - inner[i-1].z;
        double dist = std::sqrt(dx*dx + dz*dz);
        if (dist < 1e-9) return false;
    }
    return true;
}

bool BellGenerator::checkSlopeAngles() const {

}

// ============================================================
// ПРОВЕРКА КАЧЕСТВА ЭЛЕМЕНТОВ
// ============================================================
bool BellGenerator::checkElementQuality() const {
    const double min_angle_deg = 10.0;
    const double max_angle_deg = 170.0;
    const double max_aspect_ratio = 100.0;
    const double min_area = 1e-12;
    
    for (const auto& elem : mesh_data_.elements) {
        const Node& n1 = mesh_data_.nodes[elem.n1 - 1];
        const Node& n2 = mesh_data_.nodes[elem.n2 - 1];
        const Node& n3 = mesh_data_.nodes[elem.n3 - 1];
        const Node& n4 = mesh_data_.nodes[elem.n4 - 1];
        
        if (std::abs(n1.r - n2.r) < 1e-12 && std::abs(n1.z - n2.z) < 1e-12) return false;
        if (std::abs(n2.r - n3.r) < 1e-12 && std::abs(n2.z - n3.z) < 1e-12) return false;
        if (std::abs(n3.r - n4.r) < 1e-12 && std::abs(n3.z - n4.z) < 1e-12) return false;
        if (std::abs(n4.r - n1.r) < 1e-12 && std::abs(n4.z - n1.z) < 1e-12) return false;
        
        double v1x = n2.r - n1.r, v1z = n2.z - n1.z;
        double v2x = n3.r - n2.r, v2z = n3.z - n2.z;
        double v3x = n4.r - n3.r, v3z = n4.z - n3.z;
        double v4x = n1.r - n4.r, v4z = n1.z - n4.z;
        
        double l1 = std::sqrt(v1x*v1x + v1z*v1z);
        double l2 = std::sqrt(v2x*v2x + v2z*v2z);
        double l3 = std::sqrt(v3x*v3x + v3z*v3z);
        double l4 = std::sqrt(v4x*v4x + v4z*v4z);
        
        if (l1 < min_area || l2 < min_area || l3 < min_area || l4 < min_area) {
            return false;
        }
        
        double min_len = std::min({l1, l2, l3, l4});
        double max_len = std::max({l1, l2, l3, l4});
        if (max_len / min_len > max_aspect_ratio) {
            return false;
        }
        
        auto calc_angle = [](double ax, double az, double bx, double bz) -> double {
            double len_a = std::sqrt(ax*ax + az*az);
            double len_b = std::sqrt(bx*bx + bz*bz);
            if (len_a < 1e-12 || len_b < 1e-12) return 0.0;
            double cos_angle = (ax*bx + az*bz) / (len_a * len_b);
            cos_angle = std::max(-1.0, std::min(1.0, cos_angle));
            return std::acos(cos_angle) * 180.0 / M_PI;
        };
        
        double a1 = calc_angle(-v1x, -v1z, v4x, v4z);
        double a2 = calc_angle(-v2x, -v2z, v1x, v1z);
        double a3 = calc_angle(-v3x, -v3z, v2x, v2z);
        double a4 = calc_angle(-v4x, -v4z, v3x, v3z);
        
        if (a1 < min_angle_deg || a2 < min_angle_deg || a3 < min_angle_deg || a4 < min_angle_deg) {
            return false;
        }
        
        if (a1 > max_angle_deg || a2 > max_angle_deg || a3 > max_angle_deg || a4 > max_angle_deg) {
            return false;
        }
        
        double area = 0.5 * ((n1.r*n2.z + n2.r*n3.z + n3.r*n4.z + n4.r*n1.z) -
                             (n2.r*n1.z + n3.r*n2.z + n4.r*n3.z + n1.r*n4.z));
        if (area < min_area) {
            return false;
        }
    }
    return true;
}

// ============================================================
// ПОЛНАЯ ГЕОМЕТРИЧЕСКАЯ ПРОВЕРКА
// ============================================================
GeometryValidationResult BellGenerator::validateGeometry() const {
    GeometryValidationResult result;
    result.passed = true;
    
    if (!geometry_.isValid()) {
        result.passed = false;
        result.reason = "Invalid geometry parameters";
        return result;
    }
    
    if (!checkMonotonicity()) {
        result.passed = false;
        result.reason = "Non-monotonic radius knots";
        double penalty = 0.0;
        const auto& r = geometry_.r_knots;
        for (size_t i = 1; i < r.size(); ++i) {
            if (r[i] <= r[i-1]) {
                penalty += (r[i-1] - r[i]) * (r[i-1] - r[i]);
            }
        }
        result.penalty_value = penalty / (r.size() - 1);
        return result;
    }
    
    if (!mesh_data_.inner_profile.empty()) {
        if (!checkSelfIntersection()) {
            result.passed = false;
            result.reason = "Inner profile self-intersection (loops detected)";
            return result;
        }
    }
    
    return result;
}

// ============================================================
// ПОСТРОЕНИЕ СЕТКИ
// ============================================================
bool BellGenerator::buildMesh() {
    GeometryValidationResult dummy;
    return buildMesh(&dummy);
}

bool BellGenerator::buildMesh(GeometryValidationResult* validation_out) {
    mesh_data_.clear();
    is_mesh_built_ = false;
    
    auto validation = validateGeometry();
    if (validation_out) *validation_out = validation;
    if (!validation.passed) {
        return false;
    }
    
    if (!buildSpline()) {
        logError("Failed to build spline");
        return false;
    }
    
    double arc_len = computeArcLength();
    if (arc_len < 1e-6) {
        logError("Arc length is zero");
        return false;
    }
    
    computeOuterProfile();
    
    std::vector<double> n_r, n_z;
    computeNormals(n_r, n_z);
    computeInnerProfile(n_r, n_z);
    
    if (!checkSelfIntersection()) {
        logError("Self-intersection detected in inner profile");
        if (validation_out) {
            validation_out->passed = false;
            validation_out->reason = "Inner profile self-intersection (loops)";
        }
        return false;
    }
    
    assembleMesh();
    computeMetrics();
    
    if (!checkElementQuality()) {
        logError("Element quality check failed: degenerate elements detected");
        if (validation_out) {
            validation_out->passed = false;
            validation_out->reason = "Degenerate elements in mesh";
        }
        return false;
    }
    
    is_mesh_built_ = true;
    return true;
}

// ============================================================
// СОХРАНЕНИЕ В ФАЙЛ
// ============================================================
bool BellGenerator::saveToFile(const std::string& filename) const {
    if (!is_mesh_built_ || mesh_data_.nodes.empty()) {
        logError("Mesh not built or empty");
        return false;
    }
    
    std::ofstream file(filename);
    if (!file.is_open()) {
        logError("Cannot open file: " + filename);
        return false;
    }
    
    file << "*HEADING\n";
    file << "Bell - axisymmetric mesh generated by BellGenerator (GSL)\n";
    file << "*MATERIAL, NAME=BRONZE\n";
    file << "*ELASTIC\n";
    file << material_.E << ", " << material_.nu << "\n";
    file << "*DENSITY\n";
    file << material_.rho << "\n";
    
    file << "*NODE, NSET=Nall\n";
    for (const auto& node : mesh_data_.nodes) {
        file << node.id << ", " << node.r << ", " << node.z << "\n";
    }
    
    file << "*ELEMENT, TYPE=CAX4, ELSET=Eall\n";
    for (size_t i = 0; i < mesh_data_.elements.size(); ++i) {
        const auto& e = mesh_data_.elements[i];
        file << (i + 1) << ", " << e.n1 << ", " << e.n2 << ", " << e.n3 << ", " << e.n4 << "\n";
    }
    
    file << "*SOLID SECTION, ELSET=Eall, MATERIAL=BRONZE\n";
    file << "*BOUNDARY\n";
    file << "1, 1, 1, 0.0\n";
    file << (config_.nNodes + 1) << ", 1, 1, 0.0\n";
    
    file << "*STEP\n";
    file << "*FREQUENCY, STORAGE=YES\n";
    file << "10\n";
    
    file << "*NODE FILE, OUTPUT=2D\n";
    file << "U\n";
    
    file << "*EL FILE, OUTPUT=2D\n";
    file << "MASS\n";
    
    file << "*END STEP\n";
    
    file.close();
    return true;
}

// ============================================================
// СТРУКТУРА ДЛЯ СОХРАНЕНИЯ РЕЗУЛЬТАТА
// ============================================================
struct BestResult {
    double error = 0.0;
    BellGeometry geometry;
    CCXData ccx_data;
    
    bool save(const std::string& filename) const {
        json data;
        
        data["error"] = error;
        
        data["geometry"]["H"] = geometry.H;
        data["geometry"]["R_mouth"] = geometry.R_mouth;
        data["geometry"]["t_top"] = geometry.t_top;
        data["geometry"]["t_bottom"] = geometry.t_bottom;
        data["geometry"]["r_knots"] = geometry.r_knots;
        
        if (ccx_data.modes.size() > 0) {
            data["frequencies"]["hum"] = ccx_data.modes[0].frequency_hz;
        }
        if (ccx_data.modes.size() > 1) {
            data["frequencies"]["prime"] = ccx_data.modes[1].frequency_hz;
        }
        if (ccx_data.modes.size() > 2) {
            data["frequencies"]["tertia"] = ccx_data.modes[2].frequency_hz;
        }
        if (ccx_data.modes.size() > 3) {
            data["frequencies"]["quinta"] = ccx_data.modes[3].frequency_hz;
        }
        if (ccx_data.modes.size() > 4) {
            data["frequencies"]["nominal"] = ccx_data.modes[4].frequency_hz;
        }
        
        std::ofstream file(filename);
        if (!file.is_open()) return false;
        
        file << data.dump(4);
        file.close();
        return true;
    }
    
    bool load(const std::string& filename) {
        std::ifstream file(filename);
        if (!file.is_open()) return false;
        
        try {
            json data = json::parse(file);
            
            error = data.value("error", 0.0);
            
            if (data.contains("geometry")) {
                auto& g = data["geometry"];
                geometry.H = g.value("H", 1000.0);
                geometry.R_mouth = g.value("R_mouth", 400.0);
                geometry.t_top = g.value("t_top", 10.0);
                geometry.t_bottom = g.value("t_bottom", 60.0);
                
                if (g.contains("r_knots") && g["r_knots"].is_array()) {
                    geometry.r_knots.clear();
                    for (auto& item : g["r_knots"]) {
                        geometry.r_knots.push_back(item.get<double>());
                    }
                }
            }
            
            file.close();
            return true;
        } catch (const std::exception& e) {
            fmt::print(stderr, "❌ Ошибка загрузки JSON: {}\n", e.what());
            file.close();
            return false;
        }
    }
};

// ============================================================
// ЗАГРУЗКА ИЗ ФАЙЛА С ПЕРЕСЧЁТОМ ПОД НОВУЮ ЦЕЛЕВУЮ ЧАСТОТУ
// ============================================================
bool EvolutionaryOptimizer::loadBestFromFile(const std::string& filename) {
    BestResult result;
    if (!result.load(filename)) {
        return false;
    }
    
    BellGeometry loaded_geometry = result.geometry;
    
    fmt::print("📂 Loaded geometry from {}\n", filename);
    fmt::print("   Old H = {:.1f}, R_mouth = {:.1f}\n", 
               loaded_geometry.H, loaded_geometry.R_mouth);
    fmt::print("   Old error = {:.6f}\n", result.error);
    fmt::print("   🔄 Rescaling to target frequency: {:.1f} Hz\n", 
               targets_.prime_freq_target);
    
    // ============================================================
    // 1. РАСЧИТЫВАЕМ ЗАГРУЖЕННУЮ ГЕОМЕТРИЮ
    // ============================================================
    BellMeshConfig config;
    config.nNodes = 15;
    BellGenerator generator(material_, config);
    generator.setGeometry(loaded_geometry);
    
    if (!generator.buildMesh()) {
        fmt::print("   ❌ Failed to build mesh\n");
        return false;
    }
    
    if (!generator.saveToFile("bell.inp")) {
        fmt::print("   ❌ Failed to save bell.inp\n");
        return false;
    }
    
#ifdef _WIN32
    int ccx_result = std::system("ccx bell > nul 2>&1");
#else
    int ccx_result = std::system("ccx bell > /dev/null 2>&1");
#endif
    if (ccx_result != 0) {
        fmt::print("   ❌ CalculiX failed\n");
        return false;
    }
    
    CCXData first_data = BellOptimizer::CCXParser::parse_file("bell.dat", true);
    if (first_data.modes.empty()) {
        fmt::print("   ❌ No modes found\n");
        return false;
    }
    
    double prime_freq = 0.0;
    if (first_data.modes.size() > 1) {
        prime_freq = first_data.modes[1].frequency_hz;
    }
    
    if (prime_freq < 1.0) {
        fmt::print("   ❌ Invalid prime frequency: {:.2f}\n", prime_freq);
        return false;
    }
    
    fmt::print("   Current Prime: {:.2f} Hz\n", prime_freq);
    
    // ============================================================
    // 2. МАСШТАБИРУЕМ ПОД НОВУЮ ЦЕЛЕВУЮ ЧАСТОТУ
    // ============================================================
    double scale = prime_freq / targets_.prime_freq_target;
    
    if (scale < 0.05 || scale > 20.0) {
        fmt::print("   ❌ Invalid scale: {:.3f}\n", scale);
        return false;
    }
    
    fmt::print("   Scale factor: {:.3f}\n", scale);
    
    BellGeometry scaled_geometry = loaded_geometry;
    scaled_geometry.scale(scale);
    
    // ============================================================
    // 3. ПРОВЕРКИ ПОСЛЕ МАСШТАБИРОВАНИЯ (ТОЛЬКО ПРОПОРЦИИ)
    // ============================================================
    
    double t_top_ratio = scaled_geometry.t_top / scaled_geometry.H;
    double t_bottom_ratio = scaled_geometry.t_bottom / scaled_geometry.H;
    
    if (t_top_ratio < 0.005 || t_top_ratio > 0.050) {
        fmt::print("   ❌ t_top/H ratio out of range: {:.4f}\n", t_top_ratio);
        return false;
    }
    if (t_bottom_ratio < 0.010 || t_bottom_ratio > 0.100) {
        fmt::print("   ❌ t_bottom/H ratio out of range: {:.4f}\n", t_bottom_ratio);
        return false;
    }
    
    double diameter = 2.0 * scaled_geometry.R_mouth;
    double ratio_diam = diameter / scaled_geometry.H;
    if (ratio_diam < 0.7 || ratio_diam > 1.3) {
        fmt::print("   ❌ Diameter/Height ratio out of range: {:.3f}\n", ratio_diam);
        return false;
    }
    
    fmt::print("   Scaled H = {:.1f}, R_mouth = {:.1f}\n", scaled_geometry.H, scaled_geometry.R_mouth);
    fmt::print("   t_top/H = {:.4f}, t_bottom/H = {:.4f}\n", t_top_ratio, t_bottom_ratio);
    
    // ============================================================
    // 4. ПЕРЕСЧИТЫВАЕМ МАСШТАБИРОВАННУЮ ГЕОМЕТРИЮ
    // ============================================================
    generator.setGeometry(scaled_geometry);
    
    if (!generator.buildMesh()) {
        fmt::print("   ❌ Failed to build mesh for scaled geometry\n");
        return false;
    }
    
    if (!generator.saveToFile("bell.inp")) {
        fmt::print("   ❌ Failed to save bell.inp for scaled geometry\n");
        return false;
    }
    
#ifdef _WIN32
    ccx_result = std::system("ccx bell > nul 2>&1");
#else
    ccx_result = std::system("ccx bell > /dev/null 2>&1");
#endif
    if (ccx_result != 0) {
        fmt::print("   ❌ CalculiX failed for scaled geometry\n");
        return false;
    }
    
    CCXData scaled_data = BellOptimizer::CCXParser::parse_file("bell.dat", true);
    if (scaled_data.modes.empty()) {
        fmt::print("   ❌ No modes found for scaled geometry\n");
        return false;
    }
    
    // ============================================================
    // 5. СОХРАНЯЕМ ПЕРЕСЧИТАННЫЕ ДАННЫЕ
    // ============================================================
    setInitialGuess(scaled_geometry);
    best_error_ = computeError(scaled_data, scaled_geometry);
    best_geometry_ = scaled_geometry;
    best_ccx_data_ = scaled_data;
    
    fmt::print("✅ Rescaled successfully!\n");
    fmt::print("   New error: {:.6f}\n", best_error_);
    fmt::print("   New Prime: {:.2f} Hz\n", best_ccx_data_.modes[1].frequency_hz);
    
    saveBestResult();
    
    return true;
}

// ============================================================
// КОПИРОВАНИЕ ФАЙЛА
// ============================================================
void EvolutionaryOptimizer::copyFile(const std::string& src, const std::string& dst) {
    std::ifstream src_file(src, std::ios::binary);
    if (!src_file.is_open()) return;
    std::ofstream dst_file(dst, std::ios::binary);
    if (!dst_file.is_open()) return;
    dst_file << src_file.rdbuf();
}

// ============================================================
// СОХРАНЕНИЕ ЛУЧШЕГО РЕЗУЛЬТАТА
// ============================================================
void EvolutionaryOptimizer::saveBestResult() {
    fmt::print("   🔍 Saving best result...\n");
    fmt::print("      Error: {:.6f}\n", best_error_);
    fmt::print("      Modes: {}\n", best_ccx_data_.modes.size());
    
    if (best_error_ >= 1e8) {
        fmt::print("      ❌ Error too high (>= 1e8)\n");
        return;
    }
    
    if (best_ccx_data_.modes.size() < 3) {
        fmt::print("      ❌ Not enough modes: {} < 3\n", best_ccx_data_.modes.size());
        return;
    }
    
    bool has_valid_freq = false;
    for (const auto& mode : best_ccx_data_.modes) {
        if (mode.frequency_hz > 20.0) {
            has_valid_freq = true;
            break;
        }
    }
    
    if (!has_valid_freq) {
        fmt::print("      ❌ No valid frequencies (> 20 Hz)\n");
        return;
    }
    
    BestResult result;
    result.error = best_error_;
    result.geometry = best_geometry_;
    result.ccx_data = best_ccx_data_;
    
    fmt::print("      💾 Writing to best_geometry.json...\n");
    if (result.save("best_geometry.json")) {
        fmt::print("      ✅ Saved successfully!\n");
    } else {
        fmt::print("      ❌ SAVE FAILED! Cannot open file for writing.\n");
        fmt::print("      ⚠️  Check file permissions or if file is open in another program.\n");
    }
    
    copyFile("bell.inp", "best_bell.inp");
    copyFile("bell.dat", "best_bell.dat");
    fmt::print("      ✅ Copied bell.inp -> best_bell.inp, bell.dat -> best_bell.dat\n");
}

// ============================================================
// ГЕНЕРАЦИЯ МОНОТОННЫХ РАДИУСОВ
// ============================================================
std::vector<double> EvolutionaryOptimizer::generateMonotonicRadii(double Rmouth, int num_internal) {
    std::vector<double> radii;
    radii.reserve(num_internal + 2);
    
    radii.push_back(0.0);
    
    std::uniform_real_distribution<double> dist(0.0, 1.0);
    std::vector<double> fractions(num_internal);
    for (int i = 0; i < num_internal; ++i) {
        fractions[i] = dist(rng_);
    }
    std::sort(fractions.begin(), fractions.end());
    
    for (int i = 0; i < num_internal; ++i) {
        double min_fraction = 0.05 + i * 0.25;
        double max_fraction = 0.15 + i * 0.28;
        double fraction = min_fraction + fractions[i] * (max_fraction - min_fraction);
        radii.push_back(fraction * Rmouth);
    }
    
    radii.push_back(Rmouth);
    return radii;
}

// ============================================================
// ГЕНЕРАЦИЯ СЛУЧАЙНОЙ ГЕОМЕТРИИ
// ============================================================
BellGeometry EvolutionaryOptimizer::generateRandomGeometry() {
    std::uniform_real_distribution<double> distH(500.0, 2000.0);
    std::uniform_real_distribution<double> distR(0.0, 1.0);
    
    BellGeometry geom;
    
    geom.H = distH(rng_);
    
    double r_min = 0.4 * geom.H;
    double r_max = 0.8 * geom.H;
    geom.R_mouth = r_min + distR(rng_) * (r_max - r_min);
    
    std::uniform_real_distribution<double> distTtopRatio(0.008, 0.025);
    std::uniform_real_distribution<double> distTbottomRatio(0.03, 0.08);
    
    geom.t_top = distTtopRatio(rng_) * geom.H;
    geom.t_bottom = distTbottomRatio(rng_) * geom.H;
    
    int num_internal = evo_params_.num_knots - 2;
    std::vector<double> r_abs = generateMonotonicRadii(geom.R_mouth, num_internal);
    
    geom.r_knots.clear();
    for (double r : r_abs) {
        geom.r_knots.push_back(r / geom.R_mouth);
    }
    
    return geom;
}

// ============================================================
// СОЗДАНИЕ КАНДИДАТА ВБЛИЗИ ЗАГРУЖЕННОЙ ГЕОМЕТРИИ (АДАПТИВНЫЙ РАДИУС)
// ============================================================
EvolutionaryOptimizer::Candidate EvolutionaryOptimizer::createNearCandidate(const BellGeometry& base, double error) {
    Candidate cand;
    cand.error = 1e9;
    cand.valid = false;
    
    double max_error = 1.0;
    double scale = std::min(error / max_error, 1.0);
    double radius = 0.03 + 0.17 * scale;
    
    std::uniform_real_distribution<double> dist(-radius, radius);
    
    const int MAX_ATTEMPTS = 100000;
    
    for (int attempt = 0; attempt < MAX_ATTEMPTS; ++attempt) {
        BellGeometry geom = base;
        
        double delta_H = dist(rng_) * base.H;
        geom.H = std::max(500.0, std::min(2000.0, base.H + delta_H));
        
        double r_min = 0.4 * geom.H;
        double r_max = 0.8 * geom.H;
        double delta_R = dist(rng_) * base.R_mouth;
        geom.R_mouth = std::max(r_min, std::min(r_max, base.R_mouth + delta_R));
        
        double delta_t_top = dist(rng_) * base.t_top;
        geom.t_top = std::max(2.0, std::min(60.0, base.t_top + delta_t_top));
        
        double delta_t_bottom = dist(rng_) * base.t_bottom;
        geom.t_bottom = std::max(10.0, std::min(160.0, base.t_bottom + delta_t_bottom));
        
        for (size_t i = 1; i < geom.r_knots.size() - 1; ++i) {
            double delta_r = dist(rng_) * 0.15;
            geom.r_knots[i] = std::max(0.0, std::min(1.0, base.r_knots[i] + delta_r));
        }
        
        for (size_t i = 1; i < geom.r_knots.size(); ++i) {
            if (geom.r_knots[i] <= geom.r_knots[i-1]) {
                geom.r_knots[i] = geom.r_knots[i-1] + 0.01;
            }
            if (geom.r_knots[i] > 1.0) geom.r_knots[i] = 1.0;
        }
        if (geom.r_knots.back() != 1.0) geom.r_knots.back() = 1.0;
        
        if (!geom.isValid()) continue;
        
        BellMeshConfig config;
        config.nNodes = 15;
        BellGenerator generator(material_, config);
        generator.setGeometry(geom);
        
        auto validation = generator.validateGeometry();
        if (!validation.passed) continue;
        
        cand.geometry = geom;
        cand.valid = false;
        return cand;
    }
    
    fmt::print("   ⚠️ Warning: could not generate valid candidate after {} attempts. Using base geometry.\n", MAX_ATTEMPTS);
    cand.geometry = base;
    return cand;
}

// ============================================================
// ВЫЧИСЛЕНИЕ ОШИБКИ (ТОЛЬКО СООТНОШЕНИЯ + ШТРАФЫ)
// ============================================================
double EvolutionaryOptimizer::computeError(const CCXData& ccx_data, const BellGeometry& geometry) {
    if (ccx_data.modes.size() < 5) {
        return 1e9;
    }
    
    const auto& modes = ccx_data.modes;
    
    double hum_freq = modes[0].frequency_hz;
    double hum_amp = modes[0].amplitude;
    double prime_freq = modes[1].frequency_hz;
    
    if (hum_freq < 1.0 || prime_freq < 1.0) {
        return 1e9;
    }
    
    auto sigmoid = [](double x, double k) -> double {
        if (x < -100.0) return 0.0;
        if (x > 100.0) return 1.0;
        return 2.0 / (1.0 + std::exp(-k * x)) - 1.0;
    };
    
    double freq_error = 0.0;
    int mode_indices[4] = {0, 2, 3, 4};
    
    for (int i = 0; i < 4; ++i) {
        int idx = mode_indices[i];
        if (idx >= (int)modes.size()) {
            return 1e9;
        }
        double freq = modes[idx].frequency_hz;
        double ratio = freq / prime_freq;
        double error_raw = targets_.freq_weights[i] * std::pow(ratio - targets_.freq_ratios[i], 2.0);
        freq_error += sigmoid(error_raw, targets_.k_freq);
    }
    
    double amp_error = 0.0;
    int amp_mode_indices[4] = {1, 2, 3, 4};
    
    for (int i = 0; i < 4; ++i) {
        int idx = amp_mode_indices[i];
        if (idx >= (int)modes.size()) {
            return 1e9;
        }
        double amp = modes[idx].amplitude;
        double amp_ratio = amp / hum_amp;
        double error_raw = targets_.amp_weights[i] * std::pow(amp_ratio - targets_.amp_ratios[i], 2.0);
        amp_error += sigmoid(error_raw, targets_.k_amp);
    }
    
    double angle_penalty = 0.0;
    const double max_angle_deg = 45.0;
    const double max_slope = std::tan(max_angle_deg * M_PI / 180.0);
    
    for (size_t i = 1; i < geometry.r_knots.size(); ++i) {
        double dz = geometry.getKnotZ(i) - geometry.getKnotZ(i-1);
        double dr = (geometry.r_knots[i] - geometry.r_knots[i-1]) * geometry.R_mouth;
        if (dz < 1e-9) continue;
        double slope = std::abs(dr / dz);
        if (slope > max_slope) {
            double excess = slope - max_slope;
            angle_penalty += excess * excess;
        }
    }
    angle_penalty /= (geometry.r_knots.size() - 1);
    double angle_error = 0.5 * sigmoid(angle_penalty, 2.0);
    
    double proportion_penalty = 0.0;
    double diameter = 2.0 * geometry.R_mouth;
    double ratio_diam = diameter / geometry.H;
    
    double target_min = 0.7;
    double target_max = 1.3;
    
    if (ratio_diam < target_min) {
        double excess = (target_min - ratio_diam) / target_min;
        proportion_penalty += excess * excess;
    }
    if (ratio_diam > target_max) {
        double excess = (ratio_diam - target_max) / target_max;
        proportion_penalty += excess * excess;
    }
    
    double proportion_error = 0.5 * sigmoid(proportion_penalty, 2.0);
    
    double total_error = freq_error + amp_error + angle_error + proportion_error;
    
    return total_error;
}

// ============================================================
// ОЦЕНКА КАНДИДАТА (С ПРОПОРЦИОНАЛЬНЫМИ ПРОВЕРКАМИ)
// ============================================================
double EvolutionaryOptimizer::evaluateCandidate(BellGeometry& geometry, CCXData& out_data) {
    BellMeshConfig config;
    config.nNodes = 15;
    
    BellGenerator generator(material_, config);
    generator.setGeometry(geometry);
    
    auto validation = generator.validateGeometry();
    if (!validation.passed) {
        return 1e9;
    }
    
    if (!generator.buildMesh()) {
        return 1e9;
    }
    
    if (!generator.saveToFile("bell.inp")) {
        return 1e9;
    }
    
#ifdef _WIN32
    int ccx_result = std::system("ccx bell > nul 2>&1");
#else
    int ccx_result = std::system("ccx bell > /dev/null 2>&1");
#endif
    if (ccx_result != 0) {
        return 1e9;
    }
    
    CCXData first_data = BellOptimizer::CCXParser::parse_file("bell.dat", true);
    if (first_data.modes.empty()) {
        return 1e9;
    }
    
    double prime_freq = 0.0;
    if (first_data.modes.size() > 1) {
        prime_freq = first_data.modes[1].frequency_hz;
    }
    
    if (prime_freq < 1.0) {
        return 1e9;
    }
    
    double scale = prime_freq / targets_.prime_freq_target;
    
    if (scale < 0.05 || scale > 20.0) {
        return 1e9;
    }
    
    BellGeometry scaled_geometry = geometry;
    scaled_geometry.scale(scale);
    
    // ============================================================
    // ПРОВЕРКИ ПОСЛЕ МАСШТАБИРОВАНИЯ (ТОЛЬКО ПРОПОРЦИИ)
    // ============================================================
    
    double t_top_ratio = scaled_geometry.t_top / scaled_geometry.H;
    double t_bottom_ratio = scaled_geometry.t_bottom / scaled_geometry.H;
    
    if (t_top_ratio < 0.005 || t_top_ratio > 0.050) {
        return 1e9;
    }
    if (t_bottom_ratio < 0.010 || t_bottom_ratio > 0.100) {
        return 1e9;
    }
    
    double diameter = 2.0 * scaled_geometry.R_mouth;
    double ratio_diam = diameter / scaled_geometry.H;
    if (ratio_diam < 0.7 || ratio_diam > 1.3) {
        return 1e9;
    }
    
    // ============================================================
    // ВТОРОЙ РАСЧЁТ
    // ============================================================
    generator.setGeometry(scaled_geometry);
    
    if (!generator.buildMesh()) {
        return 1e9;
    }
    
    if (!generator.saveToFile("bell.inp")) {
        return 1e9;
    }
    
#ifdef _WIN32
    ccx_result = std::system("ccx bell > nul 2>&1");
#else
    ccx_result = std::system("ccx bell > /dev/null 2>&1");
#endif
    if (ccx_result != 0) {
        return 1e9;
    }
    
    out_data = BellOptimizer::CCXParser::parse_file("bell.dat", true);
    if (out_data.modes.empty()) {
        return 1e9;
    }
    
    geometry = scaled_geometry;
    
    return computeError(out_data, scaled_geometry);
}

// ============================================================
// ИНИЦИАЛИЗАЦИЯ ПОПУЛЯЦИИ (С СОХРАНЕНИЕМ ЭЛИТЫ)
// ============================================================
std::vector<EvolutionaryOptimizer::Candidate> EvolutionaryOptimizer::initializePopulation() {
    std::vector<Candidate> population;
    population.reserve(evo_params_.population_size);
    
    if (has_initial_guess_) {
        fmt::print("📂 Using loaded geometry as initial guess\n");
        fmt::print("   Error: {:.6f}\n", best_error_);
        
        double max_error = 1.0;
        double scale = std::min(best_error_ / max_error, 1.0);
        double radius = 0.03 + 0.17 * scale;
        fmt::print("   Adaptive search radius: {:.0f}%\n", radius * 100);
        
        Candidate initial_cand;
        initial_cand.geometry = initial_guess_;
        initial_cand.error = best_error_;
        initial_cand.ccx_data = best_ccx_data_;
        initial_cand.valid = true;
        population.push_back(initial_cand);
        
        for (int i = 1; i < evo_params_.population_size; ++i) {
            Candidate cand = createNearCandidate(initial_guess_, best_error_);
            population.push_back(cand);
        }
        
        return population;
    }
    
    fmt::print("🆕 No saved geometry. Using reasonable shapes + randoms.\n");
    auto reasonable = getReasonableGeometries();
    for (const auto& geom : reasonable) {
        Candidate cand;
        cand.geometry = geom;
        cand.error = 1e9;
        cand.valid = false;
        population.push_back(cand);
    }
    
    while ((int)population.size() < evo_params_.population_size) {
        population.push_back(createRandomCandidate());
    }
    
    return population;
}

// ============================================================
// ОЦЕНКА ПОПУЛЯЦИИ
// ============================================================
void EvolutionaryOptimizer::evaluatePopulation(std::vector<Candidate>& population) {
    for (auto& cand : population) {
        if (cand.valid) continue;
        
        CCXData ccx_data;
        double error = evaluateCandidate(cand.geometry, ccx_data);
        
        cand.error = error;
        cand.ccx_data = ccx_data;
        cand.valid = true;
    }
}

// ============================================================
// ОТБОР
// ============================================================
std::vector<EvolutionaryOptimizer::Candidate> EvolutionaryOptimizer::selectElite(const std::vector<Candidate>& population) {
    std::vector<Candidate> sorted = population;
    std::sort(sorted.begin(), sorted.end(), 
              [](const Candidate& a, const Candidate& b) {
                  return a.error < b.error;
              });
    
    int num_elite = std::min(evo_params_.num_elite, (int)sorted.size());
    std::vector<Candidate> elite(sorted.begin(), sorted.begin() + num_elite);
    
    if (!elite.empty() && elite.front().error < 1e8) {
        if (elite.front().error < best_error_) {
            fmt::print("   🆕 New best! Error: {:.6f} (was {:.6f})\n", 
                       elite.front().error, best_error_);
            best_error_ = elite.front().error;
            best_geometry_ = elite.front().geometry;
            best_ccx_data_ = elite.front().ccx_data;
            saveBestResult();
        } else {
            fmt::print("   ℹ️  No improvement. Best: {:.6f}, Current: {:.6f}\n", 
                       best_error_, elite.front().error);
        }
    }
    
    return elite;
}

// ============================================================
// СКРЕЩИВАНИЕ
// ============================================================
EvolutionaryOptimizer::Candidate EvolutionaryOptimizer::crossover(const Candidate& p1, const Candidate& p2) {
    Candidate child;
    child.geometry = p1.geometry;
    
    std::uniform_real_distribution<double> dist(0.0, 1.0);
    int point = dist(rng_) * 3;
    
    if (point == 0) {
        child.geometry.H = p2.geometry.H;
    } else if (point == 1) {
        child.geometry.R_mouth = p2.geometry.R_mouth;
    } else if (point == 2) {
        for (size_t i = 1; i < child.geometry.r_knots.size() - 1; ++i) {
            child.geometry.r_knots[i] = p2.geometry.r_knots[i];
        }
    }
    
    if (dist(rng_) > 0.5) {
        child.geometry.t_top = p2.geometry.t_top;
    }
    if (dist(rng_) > 0.5) {
        child.geometry.t_bottom = p2.geometry.t_bottom;
    }
    
    if (!child.geometry.isValid()) {
        return createRandomCandidate();
    }
    
    return child;
}

// ============================================================
// МУТАЦИЯ
// ============================================================
void EvolutionaryOptimizer::mutate(BellGeometry& geom) {
    std::uniform_real_distribution<double> dist(0.0, 1.0);
    std::uniform_real_distribution<double> distMutate(-1.0, 1.0);
    
    if (dist(rng_) < evo_params_.mutation_probability) {
        double delta = distMutate(rng_) * 0.05 * 1500.0;
        geom.H = std::max(500.0, std::min(2000.0, geom.H + delta));
    }
    
    if (dist(rng_) < evo_params_.mutation_probability) {
        double r_min = 0.4 * geom.H;
        double r_max = 0.8 * geom.H;
        double delta = distMutate(rng_) * 0.05 * (r_max - r_min);
        geom.R_mouth = std::max(r_min, std::min(r_max, geom.R_mouth + delta));
    }
    
    if (dist(rng_) < evo_params_.mutation_probability) {
        double current_ratio = geom.t_top / geom.H;
        double delta = distMutate(rng_) * 0.005;
        current_ratio = std::max(0.005, std::min(0.03, current_ratio + delta));
        geom.t_top = current_ratio * geom.H;
    }
    
    if (dist(rng_) < evo_params_.mutation_probability) {
        double current_ratio = geom.t_bottom / geom.H;
        double delta = distMutate(rng_) * 0.01;
        current_ratio = std::max(0.02, std::min(0.09, current_ratio + delta));
        geom.t_bottom = current_ratio * geom.H;
    }
    
    if (dist(rng_) < evo_params_.mutation_probability) {
        for (size_t i = 1; i < geom.r_knots.size() - 1; ++i) {
            double r_norm = geom.r_knots[i];
            double delta = distMutate(rng_) * 0.05;
            r_norm = std::max(0.0, std::min(1.0, r_norm + delta));
            geom.r_knots[i] = r_norm;
        }
    }
}

// ============================================================
// ЗАПУСК ОПТИМИЗАЦИИ
// ============================================================
void EvolutionaryOptimizer::run() {
    fmt::print("\n{:=^60}\n", "");
    fmt::print("{:=^60}\n", "  EVOLUTIONARY OPTIMIZER (Orthodox Bell Tuning)  ");
    fmt::print("{:=^60}\n", "");
    fmt::print("  Target Prime: {:.1f} Hz\n", targets_.prime_freq_target);
    fmt::print("  Population: {}\n", evo_params_.population_size);
    fmt::print("  Generations: {}\n", evo_params_.generations);
    fmt::print("  Mutation: {:.0f}%\n", evo_params_.mutation_probability * 100);
    fmt::print("{:=^60}\n\n", "");
    
    auto start_total = std::chrono::high_resolution_clock::now();
    
    std::vector<Candidate> population = initializePopulation();
    evaluatePopulation(population);
    
    auto elite = selectElite(population);
    
    if (!elite.empty() && elite.front().error < 1e8) {
        best_error_ = elite.front().error;
        best_geometry_ = elite.front().geometry;
        best_ccx_data_ = elite.front().ccx_data;
        saveBestResult();
    } else {
        fmt::print("⚠️ No valid candidates\n");
        best_geometry_ = getDefaultGeometry();
        best_error_ = 1e9;
    }
    
    printHeader();
    printGeneration(0, elite.empty() ? Candidate() : elite.front());
    
    for (int gen = 1; gen <= evo_params_.generations; ++gen) {
        fmt::print("\n--- Generation {} ---\n", gen);
        
        std::vector<Candidate> new_population;
        new_population.reserve(evo_params_.population_size);
        
        for (const auto& c : elite) {
            new_population.push_back(c);
        }
        
        while ((int)new_population.size() < evo_params_.population_size) {
            std::uniform_int_distribution<int> elite_dist(0, elite.size() - 1);
            const auto& p1 = elite[elite_dist(rng_)];
            const auto& p2 = elite[elite_dist(rng_)];
            
            Candidate child = crossover(p1, p2);
            mutate(child.geometry);
            new_population.push_back(child);
        }
        
        evaluatePopulation(new_population);
        elite = selectElite(new_population);
        
        printGeneration(gen, elite.front());
        
        if (best_error_ < 0.01) {
            fmt::print("\n🎯 Target achieved!\n");
            break;
        }
        
        population = new_population;
    }
    
    fmt::print("\n📝 Final save...\n");
    if (!elite.empty() && elite.front().error < 1e8) {
        if (elite.front().error < best_error_) {
            best_error_ = elite.front().error;
            best_geometry_ = elite.front().geometry;
            best_ccx_data_ = elite.front().ccx_data;
        }
        saveBestResult();
    }
    
    auto end_total = std::chrono::high_resolution_clock::now();
    auto duration = std::chrono::duration_cast<std::chrono::milliseconds>(end_total - start_total).count();
    
    fmt::print("\n{:=^60}\n", "");
    fmt::print("{:=^60}\n", "  OPTIMIZATION COMPLETED  ");
    fmt::print("{:=^60}\n", "");
    fmt::print("  Best error: {:.6f}\n", best_error_);
    fmt::print("  Time: {:.1f} s\n", duration / 1000.0);
    fmt::print("{:=^60}\n", "");
    
    printFinal(elite.front());
}

    fmt::print("\n");
    fmt::print("{:>4} {:>12} {:>10} {:>10} {:>10} {:>10} {:>12} {:>12} {:>12} {:>12} {:>12}\n",
               "Gen", "Error", "H", "Rm", "t_top", "t_bot", 
               "Hum", "Prime", "Tertia", "Quinta", "Nominal");
    fmt::print("{:-<120}\n", "");
}

void EvolutionaryOptimizer::printGeneration(int gen, const Candidate& best) {
    if (best.error >= 1e8) {
        fmt::print("{:>4} {:>12}\n", gen, "FAILED");
        return;
    }
    
    const auto& modes = best.ccx_data.modes;
    
    double hum_freq = 0.0, prime_freq = 0.0, tertia_freq = 0.0, quinta_freq = 0.0, nominal_freq = 0.0;
    
    if (modes.size() > 0) hum_freq = modes[0].frequency_hz;
    if (modes.size() > 1) prime_freq = modes[1].frequency_hz;
    if (modes.size() > 2) tertia_freq = modes[2].frequency_hz;
    if (modes.size() > 3) quinta_freq = modes[3].frequency_hz;
    if (modes.size() > 4) nominal_freq = modes[4].frequency_hz;
    
    fmt::print("{:>4} {:>12.4f} {:>10.1f} {:>10.1f} {:>10.1f} {:>10.1f} {:>12.1f} {:>12.1f} {:>12.1f} {:>12.1f} {:>12.1f}\n",
               gen, best.error, best.geometry.H, best.geometry.R_mouth,
               best.geometry.t_top, best.geometry.t_bottom,
               hum_freq, prime_freq, tertia_freq, quinta_freq, nominal_freq);
}

void EvolutionaryOptimizer::printFinal(const Candidate& best) {
    if (best.error >= 1e8) {
        fmt::print("\n  No valid solution found.\n");
        return;
    }
    
    fmt::print("\n  Best geometry:\n");
    fmt::print("    H         = {:.1f} mm\n", best.geometry.H);
    fmt::print("    R_mouth   = {:.1f} mm\n", best.geometry.R_mouth);
    fmt::print("    t_top     = {:.1f} mm\n", best.geometry.t_top);
    fmt::print("    t_bottom  = {:.1f} mm\n", best.geometry.t_bottom);
    fmt::print("    Radii     = ");
    for (double r : best.geometry.r_knots) {
        fmt::print("{:.3f} ", r);
    }
    fmt::print("\n");
    
    if (best.ccx_data.modes.size() >= 5) {
        fmt::print("\n  Frequencies:\n");
        const char* names[] = {"Hum", "Prime", "Tertia", "Quinta", "Nominal"};
        for (int i = 0; i < 5 && i < (int)best.ccx_data.modes.size(); ++i) {
            fmt::print("    {}: {:.2f} Hz", names[i], best.ccx_data.modes[i].frequency_hz);
            if (i > 0) {
                double ratio = best.ccx_data.modes[i].frequency_hz / best.ccx_data.modes[1].frequency_hz;
                fmt::print(" (ratio: {:.3f})", ratio);
            }
            fmt::print("\n");
        }
    }
}

// ПАРАМЕТРЫ ГЕОМЕТРИИ
// ============================================================
struct BellGeometry {
    double H = 1000.0;
    double R_mouth = 400.0;
    double t_top = 10.0;
    double t_bottom = 60.0;
    std::vector<double> r_knots;
    
    void scale(double factor) {
        H *= factor;
        R_mouth *= factor;
        t_top *= factor;
        t_bottom *= factor;
        // r_knots остаются нормированными
    }
    
    bool isValid() const {
        if (H <= 0.0) return false;
        if (R_mouth <= 0.0) return false;
        if (t_top <= 0.0) return false;
        if (t_bottom <= 0.0) return false;
        if (r_knots.size() < 3) return false;
        for (double r : r_knots) {
            if (r < 0.0 || r > 1.0) return false;
        }
        return true;
    }
    
    size_t numKnots() const { return r_knots.size(); }
    
    double getKnotZ(size_t i) const {
        if (r_knots.empty()) return 0.0;
        return i * H / (r_knots.size() - 1);
    }
};

#pragma once

#include <vector>

struct CCXModeData {
    int mode_number = 0;
    double eigenvalue = 0.0;
    double frequency_hz = 0.0;
    double amplitude = 0.0;
    double decay_time = 0.0;

    double part_x = 0.0, part_y = 0.0, part_z = 0.0;
    double part_rx = 0.0, part_ry = 0.0, part_rz = 0.0;

    double mass_x = 0.0, mass_y = 0.0, mass_z = 0.0;
    double mass_rx = 0.0, mass_ry = 0.0, mass_rz = 0.0;
    
    double effective_mass_norm = 0.0;
};

struct CCXData {
    std::vector<CCXModeData> modes;
    double fundamental_frequency = 0.0;
    std::string source_file;
    
    double sample_rate = 48000.0;
    double duration = 3.0;
    double t60_base = 2.0;
    double target_rms = 0.70;
    std::string material;

    void sort_by_frequency() {
        std::sort(modes.begin(), modes.end(),
                  [](const CCXModeData& a, const CCXModeData& b) {
                      return a.frequency_hz < b.frequency_hz;
                  });
    }

    void normalize_amplitudes(double max_amp = 1.0) {
        if (modes.empty()) return;
        double max_amp_found = 0.0;
        for (const auto& mode : modes) {
            if (mode.amplitude > max_amp_found) max_amp_found = mode.amplitude;
        }
        if (max_amp_found > 1e-10) {
            double scale = max_amp / max_amp_found;
            for (auto& mode : modes) mode.amplitude *= scale;
        }
    }

    size_t size() const { return modes.size(); }
    bool empty() const { return modes.empty(); }
};

class CCXParser {
public:
    static CCXData parse_file(const std::string& filename, bool quiet = true) {
        CCXData data;
        data.source_file = filename;

        std::ifstream file(filename);
        if (!file.is_open()) {
            if (!quiet) fmt::print(stderr, "❌ Ошибка: не удалось открыть файл {}\n", filename);
            data.source_file = "ERROR: file not found";
            return data;
        }

        if (!quiet) fmt::print("📂 Загрузка данных из {}...\n", filename);

        std::string line;
        bool found_eigenvalue_section = false;
        bool found_participation_section = false;
        bool found_effective_mass_section = false;
        
        while (std::getline(file, line)) {
            if (line.find("E I G E N V A L U E   O U T P U T") != std::string::npos) {
                if (!quiet) fmt::print("   📊 Найдена секция EIGENVALUE OUTPUT\n");
                found_eigenvalue_section = true;
                for (int i = 0; i < 5 && std::getline(file, line); ++i) {}
                
                while (std::getline(file, line)) {
                    if (line.empty() || line.find_first_not_of(" \t") == std::string::npos) break;
                    
                    std::istringstream iss(line);
                    int mode_num;
                    double eigenvalue, real_part, freq, imag_part;
                    
                    if (iss >> mode_num >> eigenvalue >> real_part >> freq >> imag_part) {
                        CCXModeData mode;
                        mode.mode_number = mode_num;
                        mode.eigenvalue = eigenvalue;
                        
                        if (freq > 0) {
                            mode.frequency_hz = freq;
                        } else if (real_part > 0) {
                            mode.frequency_hz = real_part / (2.0 * M_PI);
                        }
                        
                        if (mode.frequency_hz > 0.5) {
                            data.modes.push_back(mode);
                        }
                    }
                }
            }
            
            if (line.find("P A R T I C I P A T I O N   F A C T O R S") != std::string::npos) {
                if (!quiet) fmt::print("   📊 Найдена секция PARTICIPATION FACTORS\n");
                found_participation_section = true;
                for (int i = 0; i < 3 && std::getline(file, line); ++i) {}
                
                while (std::getline(file, line)) {
                    if (line.empty() || line.find_first_not_of(" \t") == std::string::npos) break;
                    
                    std::istringstream iss(line);
                    int mode_num;
                    double px, py, pz, prx, pry, prz;
                    
                    if (iss >> mode_num >> px >> py >> pz >> prx >> pry >> prz) {
                        for (auto& mode : data.modes) {
                            if (mode.mode_number == mode_num) {
                                mode.part_x = px;
                                mode.part_y = py;
                                mode.part_z = pz;
                                mode.part_rx = prx;
                                mode.part_ry = pry;
                                mode.part_rz = prz;
                                break;
                            }
                        }
                    }
                }
            }
            
            if (line.find("E F F E C T I V E   M O D A L   M A S S") != std::string::npos) {
                if (!quiet) fmt::print("   📊 Найдена секция EFFECTIVE MODAL MASS\n");
                found_effective_mass_section = true;
                for (int i = 0; i < 3 && std::getline(file, line); ++i) {}
                
                while (std::getline(file, line)) {
                    if (line.empty() || line.find_first_not_of(" \t") == std::string::npos) break;
                    
                    std::istringstream iss(line);
                    int mode_num;
                    double mx, my, mz, mrx, mry, mrz;
                    
                    if (iss >> mode_num >> mx >> my >> mz >> mrx >> mry >> mrz) {
                        for (auto& mode : data.modes) {
                            if (mode.mode_number == mode_num) {
                                mode.mass_x = mx;
                                mode.mass_y = my;
                                mode.mass_z = mz;
                                mode.mass_rx = mrx;
                                mode.mass_ry = mry;
                                mode.mass_rz = mrz;
                                break;
                            }
                        }
                    }
                }
            }
        }
        
        file.close();

        // Проверяем, что файл содержит данные
        if (!found_eigenvalue_section && !found_participation_section && !found_effective_mass_section) {
            if (!quiet) fmt::print(stderr, "⚠️ Предупреждение: файл не содержит секций с модальными данными\n");
            data.source_file = "WARNING: no modal data found";
            return data;
        }

        data.sort_by_frequency();

        double max_amp = 0.0;
        for (auto& mode : data.modes) {
            double amp = std::abs(mode.mass_rz);
            if (amp < 1e-10) {
                amp = std::sqrt(
                    mode.mass_x * mode.mass_x +
                    mode.mass_y * mode.mass_y +
                    mode.mass_z * mode.mass_z +
                    mode.mass_rx * mode.mass_rx +
                    mode.mass_ry * mode.mass_ry +
                    mode.mass_rz * mode.mass_rz
                );
            }
            mode.effective_mass_norm = amp;
            if (amp > max_amp) max_amp = amp;
        }
        
        if (max_amp > 1e-10) {
            for (auto& mode : data.modes) {
                mode.amplitude = mode.effective_mass_norm / max_amp;
            }
        } else {
            for (auto& mode : data.modes) {
                mode.amplitude = 1.0;
            }
        }

        if (!data.modes.empty()) {
            data.fundamental_frequency = data.modes[0].frequency_hz;
        }

        if (!quiet) fmt::print("   📊 Оставлено мод: {}\n", data.modes.size());
        return data;
    }
};
